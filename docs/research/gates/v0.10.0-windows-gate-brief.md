# This run's ownership record, <run>\roots.tsv (read with Get-RootRecord). Step 2.2 starts it only
# when GitRemotes is absent, or is one folder without this run's root; Update-SmokeRecord records
# the root when first seen live, records GitRemotes only if step 5.0 created it, and re-checks every
# recorded uid (gone or another uid: BLOCKED). Cleanup trashes only recorded nodes.
function Start-SmokeRecord {
    Assert-Run
    $f = Join-Path $G 'roots.tsv'
    if (Test-Path -LiteralPath $f) { throw "Start-SmokeRecord: $f already exists" }
    $top = @(Get-Rows /my-files | Where-Object { $_.name -ceq 'GitRemotes' -and -not $_.trashTime })
    if ($top.Count -gt 1 -or @($top | Where-Object { $_.type -cne 'folder' }).Count) { throw 'BLOCKED: /my-files/GitRemotes is not one folder; no ownership record was written' }
    if ($top.Count -eq 0) { "GitRemotes`t-`tabsent-at-step-2`t$(Get-Date -Format o)" | Set-Content -LiteralPath $f; return 'GitRemotes: ABSENT at step 2 (5.0 creates it; 6.2 trashes it only if 5.0 did)' }
    $kids = @(Get-Rows $GR)
    if (@($kids | Where-Object { $_.name -ceq $SmokeName }).Count) { throw "BLOCKED: $SMOKE already exists (a stale attempt); no ownership record was written" }
    $kids | Where-Object { $_.name -clike 'v010-*' } | ForEach-Object { "ANOTHER GATE'S OR ATTEMPT'S ROOT (read-only here): $($_.path) (uid $($_.uid))" }
    "GitRemotes`t$($top[0].uid)`tpresent-at-step-2`t$(Get-Date -Format o)" | Set-Content -LiteralPath $f
    "GitRemotes: PRESENT at step 2 (uid $($top[0].uid), $($kids.Count) child row(s)); not this gate's, never trashed"
}
function Update-SmokeRecord {
    Assert-Run
    $f = Join-Path $G 'roots.tsv'
    if (-not (Test-Path -LiteralPath $f)) { return 'NO ROOT RECORD: step 2.2 never started it, so this run owns no remote node' }
    $rec = @(Get-RootRecord); $bad = @()
    $top = @(Get-Rows /my-files | Where-Object { $_.name -ceq 'GitRemotes' -and -not $_.trashTime })
    $grs = @($rec | Where-Object { $_.name -ceq 'GitRemotes' })
    if ($grs.Count -eq 1 -and $grs[0].how -ceq 'absent-at-step-2' -and $top.Count -eq 1) {
        if (Test-Path -LiteralPath (Join-Path $G 'log\gitremotes-created.txt')) { "GitRemotes`t$($top[0].uid)`tcreated-by-5.0`t$(Get-Date -Format o)" | Add-Content -LiteralPath $f; "RECORDED: $GR (uid $($top[0].uid)), created by step 5.0" }
        else { $bad += "$GR exists, but step 5.0 of this run did not create it" }
    }
    foreach ($c in @($grs | Where-Object { $_.how -cne 'absent-at-step-2' })) { if ($top.Count -ne 1 -or $top[0].uid -cne $c.uid) { $bad += "$GR is gone or has another uid (recorded $($c.uid))" } }
    $live = @(if ($top.Count -eq 1) { Get-Rows $GR | Where-Object { $_.name -ceq $SmokeName -and -not $_.trashTime } })
    $old = @($rec | Where-Object { $_.name -ceq $SmokeName })
    if ($old.Count -and ($live.Count -ne 1 -or $live[0].uid -cne $old[0].uid)) { $bad += "$SMOKE is gone or has another uid (recorded $($old[0].uid))" }
    elseif ($old.Count) { "OK: $SMOKE uid $($old[0].uid)" }
    elseif ($live.Count -eq 1) { "$SmokeName`t$($live[0].uid)`tcreated`t$(Get-Date -Format o)" | Add-Content -LiteralPath $f; "RECORDED: $SMOKE (uid $($live[0].uid))" }
    $bad | ForEach-Object { "ROOT RECORD PROBLEM: $_ (STOP for Craig)" }
    if ($bad.Count) { throw 'BLOCKED: the ownership record does not match the account' }
}

# Step 5's BEFORE/AFTER for one remote file (v0.9.0 record 5.2, 5.4): `filesystem info` and a
# download read-back. The raw JSON names the account's address, so it stays in <run>\snap; the
# returned summary carries no address and is what the record quotes.
function Save-RefInfo([string]$Label, [string]$Path) {
    Assert-Run
    $sum = Join-Path $G "snap\info-$Label.summary.json"
    if (Test-Path -LiteralPath $sum) { throw "Save-RefInfo: $Label already exists in run $Run" }
    $raw = & proton-drive filesystem info $Path --json 2>> $errLog
    if ($LASTEXITCODE -ne 0) { throw "filesystem info $Path exited $LASTEXITCODE`n$($raw | Out-String)" }
    ($raw | Out-String) | Set-Content -LiteralPath (Join-Path $G "snap\info-$Label.raw.json")
    $j = ($raw | Out-String) | ConvertFrom-Json -DateKind String
    $d = Join-Path $G "snap\dl-$Label"
    New-Item -ItemType Directory -Path $d -ErrorAction Stop | Out-Null
    & proton-drive filesystem download $Path $d 2>> $errLog | Out-Null
    $dc = $LASTEXITCODE; $f = Join-Path $d (Split-Path -Path $Path -Leaf)
    $back = if ($dc -eq 0 -and (Test-Path -LiteralPath $f)) { [IO.File]::ReadAllText($f) } else { "(download exited $dc)" }
    $o = [pscustomobject]@{ label = $Label; uid = [string]$j.uid; revision = [string]$j.activeRevision.uid; claimedSize = $j.activeRevision.claimedSize
        activeStorageSize = $j.activeRevision.storageSize; totalStorageSize = $j.totalStorageSize; readBack = $back }
    ConvertTo-Json -InputObject $o | Set-Content -LiteralPath $sum
    $o
}

# Step 4.3, in a fresh -NoProfile process: the helper is the draft; exactly one GitProtonBackup
# module, 0.10.0, at $ModuleDir; its files are the tag's blobs after line-ending normalisation; the
# pinned import works.
function Test-InstalledRelease {
    Assert-Run
    Test-WindowsHelper
    $mods = @(Get-Module GitProtonBackup -ListAvailable)
    $mods | ForEach-Object { "module $($_.Version) at $($_.ModuleBase)" }
    $ok = ($mods.Count -eq 1) -and ([string]$mods[0].Version -ceq '0.10.0') -and ($mods[0].ModuleBase -eq $ModuleDir)
    $want = @(git -C $MainRepo ls-tree -r --name-only 'v0.10.0' -- GitProtonBackup)
    $files = @(Get-ChildItem -LiteralPath $ModuleDir -File -Recurse)
    "installed files: $($files.Count); the tag's module files: $($want.Count)"
    if ($files.Count -ne $want.Count -or $want.Count -eq 0) { $ok = $false }
    foreach ($p in $want) {
        $f = Join-Path $ModuleDir ($p.Substring('GitProtonBackup/'.Length) -replace '/', '\')
        $blob = git -C $MainRepo rev-parse "v0.10.0:$p"
        $got = if (Test-Path -LiteralPath $f) { git -C $MainRepo hash-object "--path=$p" -- $f } else { '(missing)' }
        "  $p  tag $blob  installed $got  $(if ($got -ceq $blob) { 'EQUAL' } else { 'DIFFERS' })"
        if ($got -cne $blob) { $ok = $false }
    }
    try { Import-Module GitProtonBackup -RequiredVersion 0.10.0 -ErrorAction Stop; 'Import-Module GitProtonBackup -RequiredVersion 0.10.0: ok' } catch { "Import-Module failed: $_"; $ok = $false }
    "INSTALLED RELEASE IS THE TAG'S MODULE: $(if ($ok) { 'YES' } else { 'NO' })"
}
