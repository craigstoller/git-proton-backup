# Starts a run: reads the Linux gate's run FRESH from C:\gpb010\run.txt (never from an earlier
# Windows attempt), creates C:\gpb010\win-a<Attempt> and its folders, records the Linux run in
# linux-run.txt, and names the new run in win-run.txt. Dot-source the kit again afterwards.
function Start-WinRun([int]$Attempt) {
    $lf = Join-Path $Root 'run.txt'
    $lin = if (Test-Path -LiteralPath $lf) { (Get-Content -LiteralPath $lf -Raw).Trim() } else { '' }
    if ($lin -cnotmatch '^gate-a[1-9][0-9]*$') { throw "BLOCKED: C:\gpb010\run.txt names '$lin', not a Linux gate run (ruling B.3: the Linux gate runs first)" }
    $dir = Join-Path $Root "win-a$Attempt"
    if ($Attempt -lt 1 -or (Test-Path -LiteralPath $dir)) { throw "BLOCKED: $dir already exists, or the attempt is below 1 (a re-run takes the next number, on Craig's word)" }
    New-Item -ItemType Directory -Path $dir, "$dir\log", "$dir\snap", "$dir\stage", "$dir\rollback", "$dir\w" -ErrorAction Stop | Out-Null
    Set-Content -LiteralPath "$dir\linux-run.txt" -Value $lin -NoNewline
    Set-Content -LiteralPath (Join-Path $Root 'win-run.txt') -Value "win-a$Attempt" -NoNewline
    "RUN STARTED: win-a$Attempt ($dir) at $(Get-Date -Format o), after Linux gate run $lin. Dot-source the kit again before the next command."
}

# Ownership, in <run>\roots.tsv (name, uid, how, time; one row per event; for each name the LAST row
# wins). Start-SmokeRecord (step 2.2) writes whether GitRemotes is present; Save-Creation records the
# uid a create just made, in the creating block; 5.2 appends trashed-by-5.2 after its own trash;
# Test-SmokeRecord re-checks and never adds a row: a recorded uid gone or changed, a live root or
# GitRemotes with no creation record, or a trashed-by-5.2 node live again, is BLOCKED.
function Get-OwnRecord { @(Get-RootRecord) | Group-Object name -CaseSensitive | ForEach-Object { $_.Group[-1] } }
function Start-SmokeRecord {
    Assert-Run
    $f = Join-Path $G 'roots.tsv'
    if (Test-Path -LiteralPath $f) { throw "Start-SmokeRecord: $f already exists" }
    $top = @(Get-Rows /my-files | Where-Object { $_.name -ceq 'GitRemotes' -and -not $_.trashTime })
    if ($top.Count -gt 1 -or @($top | Where-Object { $_.type -cne 'folder' }).Count) { throw 'BLOCKED: /my-files/GitRemotes is not one folder; no ownership record was written' }
    if ($top.Count -eq 0) { "GitRemotes`t-`tabsent-at-step-2`t$(Get-Date -Format o)" | Set-Content -LiteralPath $f; return 'GitRemotes: ABSENT at step 2 (4.0 creates it; 5.2 trashes it only if 4.0 recorded the creation)' }
    $kids = @(Get-Rows $GR)
    if (@($kids | Where-Object { $_.name -ceq $SmokeName }).Count) { throw "BLOCKED: $SMOKE already exists (a stale attempt); no ownership record was written" }
    "GitRemotes`t$($top[0].uid)`tpresent-at-step-2`t$(Get-Date -Format o)" | Set-Content -LiteralPath $f
    "GitRemotes: PRESENT at step 2 (uid $($top[0].uid), $($kids.Count) child row(s), $(@($kids | Where-Object { $_.name -clike 'v010-*' }).Count) of them v010-*); not this gate's, never trashed"
}
function Save-Creation([string]$Name, [string]$Parent, [string]$How) {
    Assert-Run
    $f = Join-Path $G 'roots.tsv'
    if (-not (Test-Path -LiteralPath $f)) { throw 'BLOCKED: no ownership record (step 2.2 never started it)' }
    $rows = @(Get-Rows $Parent | Where-Object { $_.name -ceq $Name -and -not $_.trashTime })
    if ($rows.Count -ne 1) { throw "BLOCKED: right after its create, $Parent lists $($rows.Count) live $Name (STOP for Craig; cleanup leaves it)" }
    "$Name`t$($rows[0].uid)`t$How`t$(Get-Date -Format o)" | Add-Content -LiteralPath $f
    "RECORDED: $Parent/$Name (uid $($rows[0].uid)), $How"
}
function Test-SmokeRecord {
    if (-not $G -or -not (Test-Path -LiteralPath (Join-Path $G 'roots.tsv'))) { return 'NO ROOT RECORD: this run owns no remote node' }
    $own = @(Get-OwnRecord); $bad = @()
    $top = @(Get-Rows /my-files | Where-Object { $_.name -ceq 'GitRemotes' -and -not $_.trashTime })
    $grRec = @($own | Where-Object { $_.name -ceq 'GitRemotes' })   # never $gr: names fold, and it would shadow $GR
    if ($grRec.Count -and $grRec[0].how -cin 'absent-at-step-2', 'trashed-by-5.2') { if ($top.Count) { $bad += "$GR exists, but this run has no live creation of it ($($grRec[0].how))" } }
    elseif ($grRec.Count -and ($top.Count -ne 1 -or $top[0].uid -cne $grRec[0].uid)) { $bad += "$GR is gone or has another uid (recorded $($grRec[0].uid), $($grRec[0].how))" }
    $live = @(if ($top.Count -eq 1) { Get-Rows $GR | Where-Object { $_.name -ceq $SmokeName -and -not $_.trashTime } })
    $rt = @($own | Where-Object { $_.name -ceq $SmokeName })
    if ($rt.Count -and $rt[0].how -ceq 'trashed-by-5.2') { if ($live.Count) { $bad += "$SMOKE exists again after 5.2 trashed it" } else { "TRASHED BY 5.2: $SMOKE (uid $($rt[0].uid)), absent as expected" } }
    elseif ($rt.Count -and ($live.Count -ne 1 -or $live[0].uid -cne $rt[0].uid)) { $bad += "$SMOKE is gone or has another uid (recorded $($rt[0].uid))" }
    elseif (-not $rt.Count -and $live.Count) { $bad += "$SMOKE exists but has no creation record (cleanup leaves it)" }
    elseif ($rt.Count) { "OK: $SMOKE uid $($rt[0].uid)" } else { "NOT CREATED: $SMOKE (nothing of this run's is under $GR)" }
    $bad | ForEach-Object { "ROOT RECORD PROBLEM: $_ (STOP for Craig)" }
    if ($bad.Count) { throw 'BLOCKED: the ownership record does not match the account; trash nothing' }
}

# Step 6.3, in a fresh -NoProfile process: the helper is the draft (throws if not); exactly one
# GitProtonBackup module, 0.10.0, at $ModuleDir; its files are the tag's blobs after line-ending
# normalisation; the pinned import works. Throws on NO.
function Test-InstalledRelease {
    Assert-Run
    Test-WindowsHelper -Require
    $mods = @(Get-Module GitProtonBackup -ListAvailable)
    $mods | ForEach-Object { "module $($_.Version) at $($_.ModuleBase)" }
    $ok = ($mods.Count -eq 1) -and ([string]$mods[0].Version -ceq '0.10.0') -and ($mods[0].ModuleBase -eq $ModuleDir)
    $want = @(git -C $MainRepo ls-tree -r --name-only 'v0.10.0' -- GitProtonBackup)
    $files = @(Get-ChildItem -LiteralPath $ModuleDir -File -Recurse)
    "installed files: $($files.Count); the tag's module files: $($want.Count)"
    if ($files.Count -ne $want.Count -or $want.Count -eq 0) { $ok = $false }
    foreach ($p in $want) {
        $f = Join-Path $ModuleDir ($p.Substring('GitProtonBackup/'.Length) -replace '/', '\')
        $blob = git -C $MainRepo rev-parse "v0.10.0:$p"
        $got = if (Test-Path -LiteralPath $f) { git -C $MainRepo hash-object "--path=$p" -- $f } else { '(missing)' }
        "  $p  tag $blob  installed $got  $(if ($got -ceq $blob) { 'EQUAL' } else { 'DIFFERS' })"
        if ($got -cne $blob) { $ok = $false }
    }
    try { Import-Module GitProtonBackup -RequiredVersion 0.10.0 -ErrorAction Stop; 'Import-Module GitProtonBackup -RequiredVersion 0.10.0: ok' } catch { "Import-Module failed: $_"; $ok = $false }
    "INSTALLED RELEASE IS THE TAG'S MODULE: $(if ($ok) { 'YES' } else { 'NO' })"
    if (-not $ok) { throw 'BLOCKED: the installed module is not the tag''s' }
}

# Steps 7.1 and R.5, in a fresh -NoProfile process: a real fleet verify with the installed module at
# -Version, judged against step 2.1's baseline by docs/releasing.md stand-in step 6's acceptance:
# exit 0, Complete, empty IncompleteReason, the same repo set, no new attention, no new finding
# (Invoke-ProtonBackupVerify's return, psm1:1675). Lines naming repos go to the private log only:
# the verify's own Write-Host findings (stream 6) and warnings (stream 3) included.
# FAIL throws. -AfterRollback (R.5) does not require the draft helper.
function Invoke-FleetVerify([string]$Version, [switch]$AfterRollback) {
    Assert-Run
    Assert-NoGpbEnv
    if ($AfterRollback) { Test-WindowsHelper } else { Test-WindowsHelper -Require }
    $base = Get-Content -LiteralPath (Join-Path $G 'log\verify-baseline.json') -Raw -ErrorAction Stop | ConvertFrom-Json
    Import-Module GitProtonBackup -RequiredVersion $Version -ErrorAction Stop
    "module: $((Get-Module GitProtonBackup).Version) from $((Get-Module GitProtonBackup).ModuleBase)"
    $t0 = Get-Date; $hostLog = Join-Path $G 'log\verify-host.txt'; $warnLog = Join-Path $G 'log\verify-warnings.txt'
    $r = Invoke-ProtonBackupVerify 6>> $hostLog 3>> $warnLog
    "verify host and warning lines (private, in log\): $(@(Get-Content -LiteralPath $hostLog, $warnLog -ErrorAction SilentlyContinue).Count)"
    $repos = @($r.Repos | ForEach-Object { [string]$_.RepoPath }); $reg = @($base.Registry | ForEach-Object { [string]$_ })
    "ExitCode=$($r.ExitCode) Complete=$($r.Complete) IncompleteReason='$($r.IncompleteReason)' Findings=$(@($r.Findings).Count) in $([int]((Get-Date) - $t0).TotalSeconds) s"
    $setDiff = @(Compare-Object $reg $repos -CaseSensitive)
    $was = @{}; foreach ($x in @($base.Attention)) { $was[[string]$x.RepoPath] = @($x.Findings) }
    $news = @(foreach ($x in @($r.Repos | Where-Object { $_.State -cne 'ok' })) {
        if (-not $was.ContainsKey([string]$x.RepoPath)) { "NEW ATTENTION: $($x.RepoPath): $(@($x.Findings) -join ' | ')" }
        else { foreach ($fd in @($x.Findings)) { if (@($was[[string]$x.RepoPath]) -cnotcontains $fd) { "NEW FINDING: $($x.RepoPath): $fd" } } }
    })
    @($setDiff | ForEach-Object { "REPO SET DIFFERS ($($_.SideIndicator)): $($_.InputObject)" }) + $news | Add-Content -LiteralPath (Join-Path $G 'log\verify-detail.txt')
    "repos verified: $($repos.Count); ok: $(@($r.Repos | Where-Object { $_.State -ceq 'ok' }).Count); registered at the baseline: $($reg.Count); set differences: $($setDiff.Count); new attention or findings: $($news.Count) (details, if any, in log\verify-detail.txt)"
    $pass = ($r.ExitCode -eq 0) -and ($r.Complete -eq $true) -and (-not $r.IncompleteReason) -and ($setDiff.Count -eq 0) -and ($repos.Count -eq $reg.Count) -and ($news.Count -eq 0)
    "VERIFY ACCEPTANCE: $(if ($pass) { 'PASS' } else { 'FAIL' })"
    if (-not $pass) { throw 'BLOCKED: the fleet verify does not meet the acceptance' }
    if (-not $AfterRollback) { "verify accepted $(Get-Date -Format o)" | Set-Content -LiteralPath (Join-Path $G 'log\verify-accepted.txt') }
}

**R.4 The consumer pin, reconciled from the actual state.** It runs in this order:
