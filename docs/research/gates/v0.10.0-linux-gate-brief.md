```powershell
# C:\gpb010\kit.ps1 - the Windows-side helpers of the v0.10.0 Linux gate.
# Dot-source it at the start of every PowerShell call:  . C:\gpb010\kit.ps1
# Nothing here writes to the account: its CLI calls are `filesystem list` and `filesystem info`.
# Every name comparison is case-sensitive (-ceq, -clike, ordinal sets): Main and main differ.
[Console]::OutputEncoding = [Text.UTF8Encoding]::new()
$G = 'C:\gpb010'
foreach ($d in 'log', 'snap', 'wsl') { New-Item -ItemType Directory -Force -Path (Join-Path $G $d) | Out-Null }
$errLog = Join-Path $G 'log\cli-stderr.txt'
$lagWait = 30   # seconds Save-Snapshot waits before re-taking a listing that shows trash lag

# Runs C:\gpb010\wsl\<Name>.sh in the distro, shows and saves its output (stderr merged), then
# prints its exit code. The script arrives through process substitution, not a pipe into
# `bash -s`, so a command inside it that reads stdin cannot swallow the rest of the script. Once
# PowerShell has substituted $Name, the string wsl.exe receives contains no `$`, so the launcher
# expands nothing. Measured 2026-10-16: CRLF lines are stripped, stdin reads EOF, and the exit
# code reaches $LASTEXITCODE.
function Invoke-GateWsl([string]$Name) {
    $out = wsl -d Ubuntu-24.04 -- bash -c "bash <(tr -d '\r' < /mnt/c/gpb010/wsl/$Name.sh) 2>&1"
    $code = $LASTEXITCODE
    $out | Set-Content -LiteralPath (Join-Path $G "log\$Name.txt")
    $out
    "EXIT $Name = $code"
}

# One `filesystem list --json` as rows: the six fields rule 1 compares (uid, name, type,
# creationTime, modificationTime, parentUid), plus path and trashTime. Timestamps stay strings.
function Get-Rows([string]$Path) {
    $raw = & proton-drive filesystem list $Path --json 2>> $errLog
    $code = $LASTEXITCODE
    if ($code -ne 0) { throw "filesystem list $Path exited $code`n$($raw | Out-String)" }
    $items = ($raw | Out-String) | ConvertFrom-Json -DateKind String
    foreach ($i in @($items)) {
        if ($null -eq $i) { continue }
        [pscustomobject]@{
            path = "$Path/$($i.name.value)"; name = [string]$i.name.value; type = [string]$i.type
            uid = [string]$i.uid; parentUid = [string]$i.parentUid
            creationTime = [string]$i.creationTime; modificationTime = [string]$i.modificationTime
            trashTime = [string]$i.trashTime
        }
    }
}

# Every row beneath Path, depth first: rule 3's full subtree enumeration.
function Get-Subtree([string]$Path) {
    foreach ($r in @(Get-Rows $Path)) {
        $r
        if ($r.type -ceq 'folder') { Get-Subtree $r.path }
    }
}

# Saves a listing (with -Recurse, a whole subtree) to C:\gpb010\snap\<Label>.json and shows it.
# Trash lag: for a few seconds a just-trashed node still appears in `list`, carrying a trashTime
# (docs/research/probes/stage1-results.json, transientTrashVisibilityInList; stage5-gate.md:536-538).
# Every push trashes its `.lock` as its last act (internal/repo/lock.go:135), so a listing taken
# right after a push can show it. So when a live row (any path outside /trash) carries a
# trashTime, this waits $lagWait seconds and re-takes the listing ONCE, and only the re-taken
# listing is saved and judged. A trashTime row that survives the re-take stays in the snapshot,
# and Compare-Snapshot flags it.
function Save-Snapshot([string]$Label, [string]$Path, [switch]$Recurse) {
    $take = { @(if ($Recurse) { Get-Subtree $Path } else { Get-Rows $Path }) }
    $rows = @(& $take)
    $lag = @($rows | Where-Object { $_.trashTime -and -not $_.path.StartsWith('/trash/') })
    if ($lag.Count -gt 0) {
        "LAG: $($lag.Count) live row(s) carry a trashTime ($(@($lag | ForEach-Object path) -join ', ')); waiting $lagWait s and re-taking $Label once"
        Start-Sleep -Seconds $lagWait
        $rows = @(& $take)
    }
    ConvertTo-Json -InputObject $rows -Depth 3 | Set-Content -LiteralPath (Join-Path $G "snap\$Label.json")
    "SNAPSHOT $Label = $Path$(if ($Recurse) { ' (whole subtree)' }): $($rows.Count) row(s)"
    $rows | Format-Table path, type, uid, modificationTime, trashTime -AutoSize | Out-String -Width 400
}

# Rule 1: compares two snapshots as row sets on the six fields, never as raw JSON. Prints every
# ADDED, REMOVED and CHANGED row (CHANGED: same uid, another field differs), then one verdict
# line. A live row carrying a trashTime is flagged as well (not in /trash, where every row has one):
# Save-Snapshot has already waited and re-taken once, so such a row is no longer lag (BLOCKED).
# A missing snapshot is an error, never an empty set.
function Compare-Snapshot([string]$Before, [string]$After) {
    foreach ($l in $Before, $After) {
        if (-not (Test-Path -LiteralPath (Join-Path $G "snap\$l.json"))) { throw "Compare-Snapshot: no snapshot named $l" }
    }
    $fields = 'uid', 'name', 'type', 'creationTime', 'modificationTime', 'parentUid'
    $read = { param($l) @(Get-Content -LiteralPath (Join-Path $G "snap\$l.json") -Raw | ConvertFrom-Json -DateKind String | Where-Object { $null -ne $_ }) }
    $a = @(& $read $Before); $b = @(& $read $After)
    $key = { param($r) ($fields | ForEach-Object { [string]$r.$_ }) -join "`t" }
    $inA = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
    $inB = [Collections.Generic.HashSet[string]]::new([StringComparer]::Ordinal)
    foreach ($r in $a) { [void]$inA.Add((& $key $r)) }
    foreach ($r in $b) { [void]$inB.Add((& $key $r)) }
    $gone = @($a | Where-Object { -not $inB.Contains((& $key $_)) })
    $new = @($b | Where-Object { -not $inA.Contains((& $key $_)) })
    $lines = @()
    foreach ($n in $new) {
        $o = @($gone | Where-Object { $_.uid -ceq $n.uid })
        if ($o.Count -eq 1) {
            $d = @($fields | Where-Object { [string]$o[0].$_ -cne [string]$n.$_ } | ForEach-Object { "$_ '$($o[0].$_)' -> '$($n.$_)'" })
            $lines += "  CHANGED $($n.path): $($d -join '; ')"
        } else { $lines += "  ADDED   $($n.path) ($($n.type), uid $($n.uid))" }
    }
    foreach ($o in $gone) {
        if (@($new | Where-Object { $_.uid -ceq $o.uid }).Count -eq 0) { $lines += "  REMOVED $($o.path) ($($o.type), uid $($o.uid))" }
    }
    foreach ($r in $b) { if ($r.trashTime -and -not $r.path.StartsWith('/trash/')) { $lines += "  LIVE ROW WITH trashTime: $($r.path) ($($r.trashTime))" } }
    if ($lines.Count -eq 0) { "COMPARE $Before -> ${After}: NO DIFFERENCE ($($a.Count) row(s))" }
    else { "COMPARE $Before -> ${After}: $($lines.Count) difference(s)"; $lines }
}

# Spec section 7's read-only Windows check: `filesystem info /my-files` with the Windows CLI.
# PASS means exit 0, a folder, and the same uid as the baseline this function recorded when
# called with -Baseline.
function Invoke-WindowsCheck([string]$Label, [switch]$Baseline) {
    $ef = Join-Path $G "log\wincheck-$Label.stderr.txt"
    $raw = & proton-drive filesystem info /my-files --json 2> $ef
    $code = $LASTEXITCODE
    $text = $raw | Out-String
    $text | Set-Content -LiteralPath (Join-Path $G "log\wincheck-$Label.txt")
    $uid = ''
    try { $j = $text | ConvertFrom-Json -DateKind String; if ([string]$j.type -ceq 'folder') { $uid = [string]$j.uid } } catch { }
    $file = Join-Path $G 'log\wincheck-baseline-uid.txt'
    if ($Baseline -and $code -eq 0 -and $uid) { Set-Content -LiteralPath $file -Value $uid }
    $want = if (Test-Path -LiteralPath $file) { (Get-Content -LiteralPath $file -Raw).Trim() } else { '' }
    $pass = ($code -eq 0) -and [bool]$uid -and ($uid -ceq $want)
    "WINDOWS CHECK ${Label}: exit=$code uid=$uid baseline=$want -> $(if ($pass) { 'PASS' } else { 'FAIL' })"
    if (-not $pass) { $text; Get-Content -LiteralPath $ef -Raw }
}

# The quiet check. Both scheduled tasks that run the Windows CLI are disabled and not running:
# ProjectsBackupDaily, and the weekly ProtonCliDriftCheck (Mondays 09:00). Concurrent Windows CLI
# starts can collide on the CLI's SQLite cache (SQLITE_BUSY, cli-0.9.0-certification.md 4.10).
# LastTaskResult 267009 (0x41301, SCHED_S_TASK_RUNNING) means a run is in flight right now.
# No backup worker and no proton-drive process may be running either.
function Test-Quiet {
    $ok = $true
    foreach ($name in 'ProjectsBackupDaily', 'ProtonCliDriftCheck') {
        $t = Get-ScheduledTask -TaskName $name -ErrorAction SilentlyContinue
        if (-not $t) {
            "${name}: not registered"
            if ($name -ceq 'ProjectsBackupDaily') { $ok = $false }
            continue
        }
        $i = $t | Get-ScheduledTaskInfo
        "${name}: State=$($t.State) LastRun=$($i.LastRunTime) LastResult=$($i.LastTaskResult) NextRun=$($i.NextRunTime)"
        if ([string]$t.State -cne 'Disabled' -or $i.LastTaskResult -eq 267009) { $ok = $false }
    }
    $workers = @(Get-CimInstance Win32_Process | Where-Object { $_.CommandLine -match 'backup-projects|ProjectsBackup\.psm1|Invoke-ProtonBackup' })
    $cli = @(Get-Process -Name proton-drive -ErrorAction SilentlyContinue)
    "backup workers: $($workers.Count); proton-drive processes: $($cli.Count)"
    if ($workers.Count -ne 0 -or $cli.Count -ne 0) { $ok = $false }
    "QUIET: $(if ($ok) { 'YES' } else { 'NO (wait, then run Test-Quiet again; do not go on)' })"
}

# Spec section 7 step 0: the helper git resolves on Windows must be the draft exe's bytes. Checked
# on the PATH a fresh shell would build (Machine, then User), as install.ps1's shadow check does.
# It SETS this process's PATH to that fresh PATH, so every git command after it in the same block
# resolves git-remote-proton exactly as checked. With -Require it throws unless the draft is the
# one and only hit, which stops the block before any git command runs.
function Test-WindowsHelper([switch]$Require) {
    $env:Path = [Environment]::GetEnvironmentVariable('Path', 'Machine') + ';' + [Environment]::GetEnvironmentVariable('Path', 'User')
    $want = (Get-FileHash -LiteralPath (Join-Path $G 'stage\git-remote-proton.exe') -Algorithm SHA256).Hash.ToLower()
    $all = @(Get-Command git-remote-proton -All -ErrorAction SilentlyContinue)
    "git-remote-proton on a fresh PATH: $($all.Count) hit(s); the draft exe's SHA-256 is $want"
    $ok = $false
    foreach ($c in $all) {
        $h = (Get-FileHash -LiteralPath $c.Source -Algorithm SHA256).Hash.ToLower()
        "  $($c.Source)  sha256 $h  $(& $c.Source --version)"
        if ($all.Count -eq 1 -and $h -ceq $want) { $ok = $true }
    }
    "WINDOWS HELPER IS THE DRAFT: $(if ($ok) { 'YES' } else { 'NO' })"
    if ($Require -and -not $ok) { throw 'BLOCKED: the draft exe is not the one and only git-remote-proton on a fresh PATH' }
}
```

**BLOCKED (rule 4).** A step is BLOCKED when any of the following happens:

**What the terminal shows.**

- **What the runner can see.** git prints its push and fetch summaries (`* [new branch]`,
  `! [remote rejected] … (<reason>)`, `[deleted]`) to stderr. The helper writes its `gpb:` and
  `git-remote-proton:` lines to stderr as well. `Invoke-GateWsl` merges stderr into what it
  shows, and pwsh blocks append `2>&1`.
- **What it cannot.** The helper's protocol `ok` and `error` lines go to git on stdout and stay
  invisible without `GIT_TRACE`. No pass condition here reads them. A refusal is read from git's
  `[remote rejected]` line, which carries the helper's reason in parentheses.
- **How lines are matched.** git's column spacing varies, so git's own lines are matched as

### Step 5: case variants, all inside the gate root (D2(a): the refusal branch)

All work is on `$R/v010-linux-case`. D2(a) is ruled, so §7's "If D2 allows them" branch does not
apply.

**How a refusal is checked.** After every push, a Windows snapshot of the whole repo subtree is
compared with the snapshot before it (rule 1). That is how a refused push is shown to make no ref
or pack write.

**Trash lag applies to every snapshot in this step.**

- **Why it arises.** Every push, refused or not, trashes its `.lock` as its last act
  [code `internal/repo/lock.go:135`]. For a few seconds after that, the `.lock` can still appear in
  a listing, carrying a `trashTime` (`docs/research/probes/stage1-results.json`,
  "transientTrashVisibilityInList"; `stage5-gate.md:536-538`).
- **What `Save-Snapshot` does about it.** When any live row carries a `trashTime`, it prints a
  `LAG:` line, waits 30 s, and re-takes the snapshot once. Only the re-taken snapshot is saved and
  compared.
- **How to judge it.** A repo-root `.lock` row with a `trashTime` is lag, not a write, and never a
  BLOCK by itself. A row with a `trashTime` that survives the re-take is flagged by
  `Compare-Snapshot` (`LIVE ROW WITH trashTime`) and is BLOCKED.

Every "exactly" and "NO DIFFERENCE" below is judged on the re-taken snapshot.

*Brief author's call: a withheld object.* 5.1 pushes a commit the remote does not yet hold (C2).
If the refusal had a bug, the push would therefore have to upload a pack, and the snapshot would
show it. This is the stage6 brief's discriminating-power lesson: a commit the remote already has
needs no pack, so the snapshot could not tell a refusal from a bug.
