# Ownership, in <run>\roots.tsv (name, uid, how, time; one row per event; for each name the LAST row
# wins). Start-SmokeRecord (step 2.2) writes whether GitRemotes is present; Save-Creation records the
# uid a create just made, in the creating block; 5.2 appends trashed-by-5.2 after its own trash;
# Test-SmokeRecord re-checks and never adds a row: a recorded uid gone or changed, a live root or
# GitRemotes with no creation record, or a trashed-by-5.2 node live again, is BLOCKED.
function Get-OwnRecord { @(Get-RootRecord) | Group-Object name -CaseSensitive | ForEach-Object { $_.Group[-1] } }
# This run's two remote roots, each with the creation its record must show: the smoke root (4.1)
# and the composition steps' root (N.1).
function Get-GateRoots {
    [pscustomobject]@{ Name = $SmokeName; Path = $SMOKE; How = 'created-by-4.1' }
    [pscustomobject]@{ Name = $NfdwName; Path = $NFDW; How = 'created-by-N.1' }
}
function Start-SmokeRecord {
    Assert-Run
    $f = Join-Path $G 'roots.tsv'
    if (Test-Path -LiteralPath $f) { throw "Start-SmokeRecord: $f already exists" }
    $top = @(Get-Rows /my-files | Where-Object { $_.name -ceq 'GitRemotes' -and -not $_.trashTime })
    if ($top.Count -gt 1 -or @($top | Where-Object { $_.type -cne 'folder' }).Count) { throw 'BLOCKED: /my-files/GitRemotes is not one folder; no ownership record was written' }
    if ($top.Count -eq 0) { "GitRemotes`t-`tabsent-at-step-2`t$(Get-Date -Format o)" | Set-Content -LiteralPath $f; return 'GitRemotes: ABSENT at step 2 (4.0 creates it; 5.2 trashes it only if 4.0 recorded the creation)' }

    if ($stale.Count) { throw "BLOCKED: $(@($stale | ForEach-Object Path) -join ' and ') already exist(s) (a stale attempt); no ownership record was written" }
    "GitRemotes`t$($top[0].uid)`tpresent-at-step-2`t$(Get-Date -Format o)" | Set-Content -LiteralPath $f
    "GitRemotes: PRESENT at step 2 (uid $($top[0].uid), $($kids.Count) child row(s), $(@($kids | Where-Object { $_.name -clike 'v011-*' }).Count) of them v011-*); not this gate's, never trashed"
}
function Save-Creation([string]$Name, [string]$Parent, [string]$How) {
    Assert-Run
    $f = Join-Path $G 'roots.tsv'
    if (-not (Test-Path -LiteralPath $f)) { throw 'BLOCKED: no ownership record (step 2.2 never started it)' }
    $rows = @(Get-Rows $Parent | Where-Object { $_.name -ceq $Name -and -not $_.trashTime })
    if ($rows.Count -ne 1) { throw "BLOCKED: right after its create, $Parent lists $($rows.Count) live $Name (STOP for Craig; cleanup leaves it)" }
    "$Name`t$($rows[0].uid)`t$How`t$(Get-Date -Format o)" | Add-Content -LiteralPath $f
    "RECORDED: $Parent/$Name (uid $($rows[0].uid)), $How"
}
function Test-SmokeRecord {
    if (-not $G -or -not (Test-Path -LiteralPath (Join-Path $G 'roots.tsv'))) { return 'NO ROOT RECORD: this run owns no remote node' }
    $own = @(Get-OwnRecord); $bad = @()
    $top = @(Get-Rows /my-files | Where-Object { $_.name -ceq 'GitRemotes' -and -not $_.trashTime })
    $grRec = @($own | Where-Object { $_.name -ceq 'GitRemotes' })   # never $gr: names fold, and it would shadow $GR
    if ($grRec.Count -and $grRec[0].how -cin 'absent-at-step-2', 'trashed-by-5.2') { if ($top.Count) { $bad += "$GR exists, but this run has no live creation of it ($($grRec[0].how))" } }
    elseif ($grRec.Count -and ($top.Count -ne 1 -or $top[0].uid -cne $grRec[0].uid)) { $bad += "$GR is gone or has another uid (recorded $($grRec[0].uid), $($grRec[0].how))" }
    $kids = @(if ($top.Count -eq 1) { Get-Rows $GR })
    foreach ($r in @(Get-GateRoots)) {
        $live = @($kids | Where-Object { $_.name -ceq $r.Name -and -not $_.trashTime })
        $rt = @($own | Where-Object { $_.name -ceq $r.Name })
        if ($rt.Count -and $rt[0].how -ceq 'trashed-by-5.2') { if ($live.Count) { $bad += "$($r.Path) exists again after 5.2 trashed it" } else { "TRASHED BY 5.2: $($r.Path) (uid $($rt[0].uid)), absent as expected" } }
        elseif ($rt.Count -and ($live.Count -ne 1 -or $live[0].uid -cne $rt[0].uid)) { $bad += "$($r.Path) is gone or has another uid (recorded $($rt[0].uid))" }
        elseif (-not $rt.Count -and $live.Count) { $bad += "$($r.Path) exists but has no creation record (cleanup leaves it)" }
        elseif ($rt.Count) { "OK: $($r.Path) uid $($rt[0].uid)" } else { "NOT CREATED: $($r.Path) (nothing of this run's by that name is under $GR)" }
    }
    $bad | ForEach-Object { "ROOT RECORD PROBLEM: $_ (STOP for Craig)" }
    if ($bad.Count) { throw 'BLOCKED: the ownership record does not match the account; trash nothing' }
}

# The escaped view: outside printable ASCII and tab, <U+XXXX>; a backslash, <BS>.
function ConvertTo-Esc([string]$s) {
    $b = [Text.StringBuilder]::new()
    foreach ($r in $s.EnumerateRunes()) {
        $v = $r.Value
        if ($v -eq 92) { [void]$b.Append('<BS>') }
        elseif ($v -eq 9 -or ($v -ge 32 -and $v -le 126)) { [void]$b.Append([char]$v) }
        else { [void]$b.Append(('<U+{0:X4}>' -f $v)) }
